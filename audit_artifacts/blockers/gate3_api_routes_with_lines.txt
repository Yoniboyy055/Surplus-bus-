    26	
    27	  return NextResponse.json({

    30	      status: successRate > 90 ? 'healthy' : successRate > 70 ? 'degraded' : 'critical',

    32	      items_queued_24h: itemsQueuedLast24h
    33	    },
    34	    recent_logs: logs

     8	  const { data: rules } = await supabase.from("alert_rules").select("id,category").eq("profile_id", user.id);

    21	    opened = (events || []).filter((e: any) => e.event_type === "opened").length;
    22	  }
    23	
    24	  const categoryCounts = new Map<string, number>();
    25	  for (const row of rules || []) {
    26	    categoryCounts.set(row.category, (categoryCounts.get(row.category) || 0) + 1);
    27	  }
    28	
    29	  return NextResponse.json({
    30	    alerts_sent_7d: sent,
    31	    alerts_opened_7d: opened,
    32	    open_rate: sent > 0 ? Math.round((opened / sent) * 100) : 0,
    33	    top_categories: [...categoryCounts.entries()].map(([category, count]) => ({ category, count })),
    34	  });
    35	}

    20	
    21	  if (!rl.allowed) {

    23	  }

    27	
    28	  if (!parsed.success) {

    30	  }
    31	
    32	  // Honeypot should drop silently
    33	  if (parsed.data.website !== undefined && parsed.data.website !== "") {
    34	    return NextResponse.json({ ok: true });
    35	  }

    40	  }
    41

    43	    const { error } = await supabase
    44	      .from("beta_signups")

   186	    .from("deals")
   187	    .update({ status })
   188	    .eq("id", dealId);
   189	
   190	  if (error) return NextResponse.json({ error: error.message }, { status: 500 });

   192	  // Audit log

   194	    deal_id: dealId,

   196	    actor_role: "operator",

     1	import { NextResponse } from "next/server";

     4	
     5	export async function GET(request: Request) {
     6	  const { supabase, user, error } = await requireUser();
     7	  if (error || !supabase || !user) return error!;
     8	
     9	  const url = new URL(request.url);
    10	  const limit = Math.min(Number(url.searchParams.get("limit") || 20), 50);
    11	
    12	  const { data: pref } = await supabase
    13	    .from("user_preferences")
    14	    .select("provinces,categories,min_value,max_value,urgency_days")
    15	    .eq("profile_id", user.id)
    16	    .maybeSingle();
    17	
    18	  const { data: rows, error: rankedError } = await supabase
    19	    .from("opportunity_features")
    20	    .select(
    21	      "normalized_score,demand_score,value_score,urgency_score,opportunity_id,opportunities(id,title,province,category,estimated_value,closing_date,source,status)"
    22	    )
    23	    .order("normalized_score", { ascending: false })
    24	    .limit(limit * 3);
    25	
    26	  if (rankedError) return NextResponse.json({ error: rankedError.message }, { status: 400 });
    27	
    28	  const provinces = pref?.provinces || [];

    31	  const maxValue = pref?.max_value ?? null;
    32	  const urgencyDays = pref?.urgency_days ?? 7;
    33	
    34	  const scored = (rows || [])
    35	    .map((row: any) => {
    36	      const opp = row.opportunities;
    37	      if (!opp) return null;
    38	
    39	      const provinceMatch = provinces.length ? Number(provinces.includes(opp.province)) : 0;
    40	      const categoryMatch = categories.length ? Number(categories.includes(opp.category)) : 0;
    41	      const valueMatch =
    42	        minValue !== null && maxValue !== null && typeof opp.estimated_value === "number"
    43	          ? Number(opp.estimated_value >= minValue && opp.estimated_value <= maxValue)
    44	          : 0;
    45	      const urgencyMatch = Number(urgencyScore(opp.closing_date) >= urgencyScore(new Date(Date.now() + urgencyDays * 86400000).toISOString()));
    46	
    47	      const userScore = personalizedScore(row.normalized_score, {
    48	        province_match: provinceMatch,
    49	        category_match: categoryMatch,
    50	        value_match: valueMatch,
    51	        urgency_match: urgencyMatch,
    52	      });
    53	
    54	      return {
    55	        id: opp.id,
    56	        title: opp.title,
    57	        source: opp.source,
    58	        province: opp.province,
    59	        category: opp.category,
    60	        estimated_value: opp.estimated_value,
    61	        closing_date: opp.closing_date,
    62	        status: opp.status,
    63	        score: userScore,
    64	        score_breakdown: {
    65	          base_normalized: row.normalized_score,
    66	          demand: row.demand_score,
    67	          value: row.value_score,
    68	          urgency: row.urgency_score,
    69	        },
    70	      };
    71	    })
    72	    .filter(Boolean)
    73	    .sort((a: any, b: any) => b.score - a.score)
    74	    .slice(0, limit);
    75	
    76	  return NextResponse.json({ opportunities: scored });
    77	}

===== app/api/opportunities/recompute-scores/route.ts =====
     1	import { NextResponse } from "next/server";
     2	import { requireOperator } from "@/lib/auth/requireUser";
     3	import { computeScore, demandScore, normalizeScore, urgencyScore, valueScore } from "@/lib/intelligence/scoring";
     4	
     5	export async function POST() {
     6	  const { supabase, error } = await requireOperator();
     7	  if (error || !supabase) return error!;
     8	
     9	  const ninetyDaysAgo = new Date(Date.now() - 90 * 24 * 60 * 60 * 1000).toISOString();
    10	
    11	  const { data: opportunities, error: oppError } = await supabase
    12	    .from("opportunities")
    13	    .select("id,category,estimated_value,closing_date,created_at")
    14	    .gte("created_at", ninetyDaysAgo)
    15	    .neq("status", "closed");
    16	
    17	  if (oppError) return NextResponse.json({ error: oppError.message }, { status: 400 });
    18	
    19	  if (!opportunities?.length) {
    20	    return NextResponse.json({ ok: true, processed: 0 });
    21	  }
    22	
    23	  const categoryCount = new Map<string, number>();
    24	  const categoryValueTotals = new Map<string, { sum: number; count: number }>();
    25	
    26	  for (const row of opportunities) {
    27	    const category = row.category || "uncategorized";
    28	    categoryCount.set(category, (categoryCount.get(category) || 0) + 1);
    29	    if (typeof row.estimated_value === "number") {
    30	      const current = categoryValueTotals.get(category) || { sum: 0, count: 0 };
    31	      categoryValueTotals.set(category, { sum: current.sum + row.estimated_value, count: current.count + 1 });
    32	    }
    33	  }
    34	
    35	  const maxCategoryCount = Math.max(...categoryCount.values(), 1);
    36	
    37	  const interim = opportunities.map((row) => {
    38	    const category = row.category || "uncategorized";
    39	    const avg = categoryValueTotals.get(category);
    40	    const avgValue = avg && avg.count > 0 ? avg.sum / avg.count : 1;
    41	
    42	    const demand = demandScore(categoryCount.get(category) || 0, maxCategoryCount);
    43	    const value = valueScore(row.estimated_value, avgValue);
    44	    const urgency = urgencyScore(row.closing_date);
    45	    const base = computeScore({ demand_score: demand, value_score: value, urgency_score: urgency });
    46	
    47	    return {
    48	      opportunity_id: row.id,
    49	      demand_score: Number(demand.toFixed(4)),
    50	      value_score: Number(value.toFixed(4)),
    51	      urgency_score: Number(urgency.toFixed(4)),
    52	      base_score: Number(base.toFixed(4)),
    53	      normalized_score: 0,
    54	      computed_at: new Date().toISOString(),
    55	    };
    56	  });
    57	
    58	  const minBase = Math.min(...interim.map((x) => x.base_score));
    59	  const maxBase = Math.max(...interim.map((x) => x.base_score));
    60	
    61	  const finalRows = interim.map((row) => ({
    62	    ...row,
    63	    normalized_score: normalizeScore(row.base_score, minBase, maxBase),
    64	  }));
    65	
    66	  const { error: upsertError } = await supabase
    67	    .from("opportunity_features")
    68	    .upsert(finalRows, { onConflict: "opportunity_id", ignoreDuplicates: false });
    69	
    70	  if (upsertError) return NextResponse.json({ error: upsertError.message }, { status: 400 });
    71	
    72	  return NextResponse.json({ ok: true, processed: finalRows.length });
    73	}
